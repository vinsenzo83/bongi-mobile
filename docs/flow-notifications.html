6DB 스키마 (3 신규 테이블)
CREATE TABLE bongi_notification_templates ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), code TEXT UNIQUE NOT NULL, -- 'WIRELESS_NEW' · 'USIM_MNO_NEW' · 'USIM_MVNO_NEW' · 'WIRED_NEW' · 'ADDON_REMIND' · 'PLAN_REMIND' category TEXT NOT NULL, -- '개통/법정' · '마케팅' · '운영' channel TEXT NOT NULL, -- 'kakao' · 'push' · 'sms' title TEXT NOT NULL, body TEXT NOT NULL, -- 변수 {{고객명}} 등 포함 variables JSONB, -- [{key:'고객명', from:'name', type:'string'}, ...] buttons JSONB, -- [{label:'채널 추가', url:'...'}] is_active BOOLEAN DEFAULT true, carrier_rules JSONB, -- {SKT:'EOMONTH(M,2)+2', KT:'...', LG:'M+100'} created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE bongi_notification_logs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), template_code TEXT REFERENCES bongi_notification_templates(code), customer_id UUID REFERENCES bongi_customers(id), source_table TEXT, -- 'incentive_sales' · 'bongi_mvno_subscriptions' 등 source_id UUID, -- 트리거 row id channel TEXT NOT NULL, -- 실제 발송 채널 recipient_phone TEXT NOT NULL, rendered_body TEXT, -- 변수 치환 후 최종 본문 status TEXT NOT NULL, -- 'queued' · 'sent' · 'delivered' · 'failed' · 'fallback_sms' provider_response JSONB, -- 채널톡/카카오 API 응답 failed_reason TEXT, scheduled_at TIMESTAMPTZ, -- D-3 cron의 경우 예약 시각 sent_at TIMESTAMPTZ, delivered_at TIMESTAMPTZ, cost_krw INTEGER, -- 발송 비용 (정산용) created_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE bongi_notification_settings ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), template_code TEXT REFERENCES bongi_notification_templates(code), trigger_type TEXT NOT NULL, -- 'instant' · 'cron' cron_expression TEXT, -- '0 9 * * *' (매일 9시) days_before INTEGER, -- D-3 · D-7 · D-1 fallback_enabled BOOLEAN DEFAULT true, -- SMS fallback on/off enabled BOOLEAN DEFAULT true, updated_at TIMESTAMPTZ DEFAULT now() ); -- 트리거 함수 (개통 즉시 발송) CREATE FUNCTION notify_on_activation() RETURNS trigger AS $$ BEGIN IF NEW.status = 'activated' AND (OLD.status IS NULL OR OLD.status != 'activated') THEN INSERT INTO bongi_notification_logs (template_code, source_table, source_id, customer_id, recipient_phone, status, scheduled_at) VALUES ('WIRELESS_NEW', TG_TABLE_NAME, NEW.id, NEW.customer_id, NEW.phone, 'queued', now()); PERFORM pg_notify('notification_queue', NEW.id::text); END IF; RETURN NEW; END; $$ LANGUAGE plpgsql;
