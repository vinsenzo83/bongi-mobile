3 · 데이터 모델 (DB 설계 초안)
-- 회원 (기존 회원 테이블 확장 또는 신규) CREATE TABLE bongi_members ( id, phone UNIQUE, verified_at, referrer_id, tier, point_balance, created_at ); -- ① 포인트 원장 (잔액은 집계 — 단일 컬럼 부채 방지) CREATE TABLE bongi_point_ledger ( id, member_id, delta /* +적립/−차감 */, reason, ref_type, ref_id, created_at ); -- ② 응모권 (월별, 소멸) CREATE TABLE bongi_tickets ( id, member_id, qty, source, period /* YYYYMM */, expires_at, created_at ); -- ③ 추천 CREATE TABLE bongi_referrals ( id, referrer_id, referee_id, status, reward_point, reward_ticket, milestone, created_at ); -- ④ 앰배서더 점수 (월별 집계) CREATE TABLE bongi_ambassador_score ( id, member_id, period, score, breakdown JSONB, rank ); -- ⑤ 등급 정의 (어드민 수정 — 하드코딩 X) CREATE TABLE bongi_member_tiers ( tier, threshold_point, benefits JSONB, active ); -- ⑥ 경품 이벤트 / 응모 / 당첨 CREATE TABLE bongi_prize_events ( id, period, budget, prizes JSONB, draw_at, status ); CREATE TABLE bongi_prize_entries ( id, event_id, member_id, ticket_qty, won_rank, payout_status, tax_withheld ); -- ⑦ AI 콘텐츠 이용 로그 (적립·어뷰징 추적) CREATE TABLE bongi_ai_content_log ( id, member_id, content_type, used_at, point_awarded, device_fp, ip ); -- ⑧ 판매점 추천 CREATE TABLE bongi_store_referral ( id, store_id, member_id, ref_type, points, created_at ); -- 적립 정책 (어드민 관리 — 적립값 하드코딩 금지, NO_HARDCODE 룰) CREATE TABLE bongi_point_rules ( reason PK, point, daily_cap, active );
설계 포인트: 포인트 잔액은 원장(ledger) 합계로 산출(단일 balance 컬럼만 두면 정합성·감사 불가). 적립값은 bongi_point_rules에서 어드민 관리(하드코딩 X). 모든 적립에 device_fp·ip 기록(어뷰징 추적).
